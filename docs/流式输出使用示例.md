```go
func (s *genkitService) ChatStream(ctx context.Context, req *model.ChatRequest) (<-chan *model.TencentCloudStreamMessage, error) {
    outputChan := make(chan *model.TencentCloudStreamMessage, 10)

                Message: "",
            },

            IsStop:       false,
        }
        
        // 发送完成事件
        outputChan <- &model.TencentCloudStreamMessage{
            CompletionID: sessionID,
            SessionID:    sessionID,
            Processes: model.ProcessInfo{
                Stage:   model.StreamStageOutput,
                Message: "",
            },
            Content:      "你好，我是AI助手",
            FinishReason: "stop",
            IsStop:       true,
        }
    }()
    
    return outputChan, nil
}
```

### JavaScript/TypeScript

```typescript
// 定义类型
interface TencentCloudStreamMessage {
  completion_id: string;
  session_id?: string;
  processes: {
    stage: string;
    message: string;
    delta_content?: string;
    content?: string;
    detail?: any;
  };
  delta_content: string;
  content: string;
  finish_reason: string;
  is_stop: boolean;
  answer_source?: string;
  additional_content?: {
    reference_chunks?: any[];
    reference_docs?: any[];
    scenario?: string;
  };
}

// 创建 EventSource
const eventSource = new EventSource('/api/v1/chat/sessions/xxx/messages/stream', {
  method: 'POST',
  headers: {
    'Content-Type': 'application/json',
    'Authorization': `Bearer ${token}`
  },
  body: JSON.stringify({
    message: '你好'
  })
});

// 处理消息
eventSource.addEventListener('message', (event) => {
  const data: TencentCloudStreamMessage = JSON.parse(event.data);
  
  // 根据阶段处理
  switch (data.processes.stage) {
    case 'tool_call_start':
      console.log('工具调用开始:', data.processes.message);
      showToolCallStatus(data.processes.detail);
      break;
      
    case 'tool_call_complete':
      console.log('工具调用完成:', data.processes.detail);
      break;
      
    case 'internal_searching':
      console.log('正在搜索:', data.processes.message);
      showSearchStatus(data.processes.detail);
      break;
      
    case 'finished_internal_searching':
      console.log('搜索完成:', data.processes.message);
      showSearchResults(data.additional_content?.reference_chunks);
      break;
      
    case 'thinking':
      // 显示思考过程
      appendThinking(data.processes.delta_content);
      break;
      
    case '':
      // 正式输出
      appendContent(data.delta_content);
      break;
  }
  
  // 检查是否完成
  if (data.is_stop) {
    eventSource.close();
    onStreamComplete(data);
  }
});

// 处理 finish 事件
eventSource.addEventListener('finish', (event) => {
  const data: TencentCloudStreamMessage = JSON.parse(event.data);
  console.log('流式输出完成');
  console.log('完整内容:', data.content);
  console.log('引用文档:', data.additional_content?.reference_docs);
  eventSource.close();
});

// 错误处理
eventSource.onerror = (error) => {
  console.error('流式输出错误:', error);
  eventSource.close();
};
```

### React 示例

```tsx
import { useState, useEffect } from 'react';

interface Message {
  role: 'user' | 'assistant';
  content: string;
  thinking?: string;
  references?: any[];
}

function ChatComponent() {
  const [messages, setMessages] = useState<Message[]>([]);
  const [currentMessage, setCurrentMessage] = useState('');
  const [currentThinking, setCurrentThinking] = useState('');
  const [isStreaming, setIsStreaming] = useState(false);
  
  const sendMessage = async (text: string) => {
    // 添加用户消息
    setMessages(prev => [...prev, { role: 'user', content: text }]);
    
    // 开始流式接收
    setIsStreaming(true);
    setCurrentMessage('');
    setCurrentThinking('');
    
    const eventSource = new EventSource(
      `/api/v1/chat/sessions/${sessionId}/messages/stream`,
      {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${token}`
        },
        body: JSON.stringify({ message: text })
      }
    );
    
    eventSource.addEventListener('message', (event) => {
      const data = JSON.parse(event.data);
      
      switch (data.processes.stage) {
        case 'thinking':
          setCurrentThinking(prev => prev + data.processes.delta_content);
          break;
          
        case '':
          setCurrentMessage(prev => prev + data.delta_content);
          break;
      }
      
      if (data.is_stop) {
        setMessages(prev => [...prev, {
          role: 'assistant',
          content: data.content,
          thinking: currentThinking,
          references: data.additional_content?.reference_docs
        }]);
        setCurrentMessage('');
        setCurrentThinking('');
        setIsStreaming(false);
        eventSource.close();
      }
    });
    
    eventSource.onerror = () => {
      setIsStreaming(false);
      eventSource.close();
    };
  };
  
  return (
    <div>
      {messages.map((msg, i) => (
        <div key={i} className={msg.role}>
          {msg.thinking && (
            <div className="thinking">{msg.thinking}</div>
          )}
          <div className="content">{msg.content}</div>
          {msg.references && (
            <div className="references">
              {msg.references.map((ref, j) => (
                <a key={j} href={ref.url}>{ref.title}</a>
              ))}
            </div>
          )}
        </div>
      ))}
      
      {isStreaming && (
        <div className="assistant streaming">
          {currentThinking && (
            <div className="thinking">{currentThinking}</div>
          )}
          <div className="content">{currentMessage}</div>
        </div>
      )}
    </div>
  );
}
```

### Python 示例

```python
import requests
import json

def stream_chat(session_id: str, message: str, token: str):
    url = f'http://localhost:8080/api/v1/chat/sessions/{session_id}/messages/stream'
    headers = {
        'Authorization': f'Bearer {token}',
        'Content-Type': 'application/json'
    }
    data = {'message': message}
    
    with requests.post(url, headers=headers, json=data, stream=True) as response:
        full_content = ''
        thinking_content = ''
        
        for line in response.iter_lines():
            if not line:
                continue
                
            line = line.decode('utf-8')
            
            # 解析 SSE 格式
            if line.startswith('event:'):
                event_type = line[6:].strip()
                continue
            
            if line.startswith('data:'):
                data_str = line[5:].strip()
                data = json.loads(data_str)
                
                # 处理不同阶段
                stage = data['processes']['stage']
                
                if stage == 'tool_call_start':
                    print(f"工具调用: {data['processes']['message']}")
                    
                elif stage == 'internal_searching':
                    print(f"搜索中: {data['processes']['message']}")
                    
                elif stage == 'finished_internal_searching':
                    print(f"搜索完成: {data['processes']['message']}")
                    if data.get('additional_content', {}).get('reference_chunks'):
                        print(f"找到 {len(data['additional_content']['reference_chunks'])} 个引用")
                        
                elif stage == 'thinking':
                    thinking_content += data['processes'].get('delta_content', '')
                    print(f"思考: {thinking_content}", end='\r')
                    
                elif stage == '':
                    # 正式输出
                    full_content += data.get('delta_content', '')
                    print(full_content, end='', flush=True)
                
                # 检查是否完成
                if data.get('is_stop'):
                    print('\n\n完成!')
                    if data.get('additional_content', {}).get('reference_docs'):
                        print('引用文档:')
                        for doc in data['additional_content']['reference_docs']:
                            print(f"  - {doc['title']}: {doc['url']}")
                    break
```

## 常见问题

### Q1: 如何区分思考过程和正式输出？

A: 通过 `processes.stage` 字段：

- `thinking`: 思考过程，内容在 `processes.delta_content`
- `""` (空字符串): 正式输出，内容在 `delta_content`

### Q2: 如何获取完整内容？

A: 在 finish 事件中，`content` 字段包含完整内容。或者客户端累积所有 `delta_content`。

### Q3: 如何处理引用文档？

A: 在 `finished_internal_searching` 阶段的 `additional_content.reference_chunks` 中获取引用片段，在 finish 事件的 `additional_content.reference_docs` 中获取引用文档列表。

### Q4: 如何判断流结束？

A: 检查 `is_stop` 字段为 `true`，或者接收到 `event: finish` 事件。

### Q5: 如何处理错误？

A: 当 `finish_reason` 为 `"error"` 时表示出错，错误信息在 `processes.message` 中。
